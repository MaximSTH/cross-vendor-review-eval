def load_cdi_config(spec_dirs: list[str]) -> CDI_RETURN_TYPE | None:
    # Load and merge devices from all YAML and JSON CDI configuration files.
    # The directory order supplied by the caller takes precedence, followed by
    # alphabetical traversal within each directory.
    devices: list[CDI_DEVICE] = []
    device_names: set[str] = set()

    for spec_dir in spec_dirs:
        for root, dirs, files in os.walk(spec_dir):
            dirs.sort()
            for file in sorted(files):
                _, ext = os.path.splitext(file)
                file_path = os.path.join(root, file)
                if ext in [".yaml", ".yml"]:
                    try:
                        with open(file_path, "r") as stream:
                            cdi = yaml.safe_load(stream)
                    except (OSError, yaml.YAMLError) as e:
                        logger.warning(f"Failed to load YAML file {file_path}: {e}")
                        continue
                elif ext == ".json":
                    try:
                        with open(file_path, "r") as stream:
                            cdi = json.load(stream)
                    except (OSError, json.JSONDecodeError, UnicodeDecodeError) as e:
                        logger.warning(f"Failed to load JSON file {file_path}: {e}")
                        continue
                else:
                    continue

                try:
                    cdi_devices = cdi.get("devices", [])
                    for device in cdi_devices:
                        name = device["name"]
                        if name not in device_names:
                            devices.append(device)
                            device_names.add(name)
                except (AttributeError, KeyError, TypeError) as e:
                    logger.warning(f"Unable to process CDI configuration {file_path}: {e}")

    return {"devices": devices} if devices else None

    if os.path.exists('/proc/device-tree/compatible'):
